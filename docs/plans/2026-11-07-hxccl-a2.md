# hxccl A2 (NCCL/RCCL API 전체, CPU 버퍼) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** NCCL v2.32.3 / RCCL rocm-7.2.4 API 중 CPU 버퍼와 MPI만으로 의미가 성립하는 것을 전부 hxccl에 구현한다.

**Architecture:** 공개 함수는 인자를 즉시 검사하고, 실행을 클로저로 만들어 `dispatch`에 넘긴다. 그룹 밖이면 바로 실행하고 그룹 안이면 큐에 쌓는다. 리덕션 op는 호출 시점에 `OpPlan`(기본 op + 입력 사전 곱 + 결과 정수 나눗셈)으로 풀어서, 알고리즘은 A와 같이 기본 op 4개만 안다.

**Tech Stack:** C++17, OpenMPI 5.0.11(conda env `hxccl`), CMake 4.4.5 + Ninja, ctest.

**Spec:** `docs/specs/2026-11-07-hxccl-a2-api-design.md`

## Global Constraints

- 결과 코드: `Success 0, UnhandledCudaError 1, SystemError 2, InternalError 3, InvalidArgument 4, InvalidUsage 5, RemoteError 6, InProgress 7, Timeout 8, NumResults 9`.
- dtype: `Int8 0, Uint8 1, Int32 2, Uint32 3, Int64 4, Uint64 5, Float16 6, Float32 7, Float64 8, Bfloat16 9, Float8e4m3 10, Float8e5m2 11, NumTypes 12`(별칭 `Char 0, Int 2, Half 6, Float 7, Double 8`).
- op: `Sum 0, Prod 1, Max 2, Min 3, Avg 4, NumOps 5`, 사용자 op는 `NumOps + 인덱스`.
- 버전 매크로: `(X <= 2 && Y <= 8) ? X*1000 + Y*100 + Z : X*10000 + Y*100 + Z`, hxccl 0.2.0 → 200.
- 인자 오류는 아무것도 보내기 전에 반환. MPI 오류는 `SystemError`. MPI 메시지 하나는 INT_MAX 바이트 이하이고, 넘으면 모든 rank가 같은 판단으로 거절한다.
- 태그: collective 0, Send/Recv 1.
- 빌드·테스트: `cmake -S . -B build -G Ninja && cmake --build build && ctest --test-dir build --output-on-failure`(conda env `hxccl` 활성화 상태). 테스트마다 `mpirun --oversubscribe -np {1,2,3,4,5,8}`.

```cpp
#pragma once
#include <cmath>
#include <cstdint>

// Software float16 / bfloat16 / fp8 (OCP) conversions, round-to-nearest-even.
// IEEE: all-ones exponent is inf/NaN. Otherwise (e4m3fn) only S.1111.111 is NaN.
namespace fp {

template <int E, int M, bool IEEE>
float decode(uint32_t v) {
  const int bias = (1 << (E - 1)) - 1;
  const uint32_t emask = (1u << E) - 1, mmask = (1u << M) - 1;
  const uint32_t e = (v >> M) & emask, m = v & mmask;
  const float sign = ((v >> (E + M)) & 1) ? -1.0f : 1.0f;
  if (IEEE && e == emask) return m ? NAN : sign * INFINITY;
  if (!IEEE && e == emask && m == mmask) return NAN;
  if (e == 0) return sign * std::ldexp(static_cast<float>(m), 1 - bias - M);
  return sign * std::ldexp(static_cast<float>(m | (1u << M)), static_cast<int>(e) - bias - M);
}

template <int E, int M, bool IEEE>
uint32_t encode(float x) {
  const int bias = (1 << (E - 1)) - 1, emin = 1 - bias;
  const uint32_t emask = (1u << E) - 1, sbit = 1u << (E + M);
  const uint32_t nan = IEEE ? (emask << M) | (1u << (M - 1)) : (emask << M) | ((1u << M) - 1);
  if (std::isnan(x)) return nan;
  const uint32_t s = std::signbit(x) ? sbit : 0;
  const uint32_t overflow = IEEE ? s | (emask << M) : s | nan;
  const float a = std::fabs(x);
  if (std::isinf(a)) return overflow;
  if (a == 0.0f) return s;
  const int emax = IEEE ? static_cast<int>(emask) - 1 - bias : static_cast<int>(emask) - bias;
  const float maxv = std::ldexp(static_cast<float>(IEEE ? (2u << M) - 1 : (2u << M) - 2),
                                emax - M);
  const int scale = std::max(std::ilogb(a), emin) - M;
  const float v = std::ldexp(std::nearbyint(std::ldexp(a, -scale)), scale);
  if (v > maxv) return overflow;
  if (v == 0.0f) return s;
  const int ve = std::ilogb(v);
  if (ve < emin) return s | static_cast<uint32_t>(std::ldexp(v, M - emin));
  const uint32_t mant = static_cast<uint32_t>(std::ldexp(v, M - ve)) - (1u << M);
  return s | (static_cast<uint32_t>(ve + bias) << M) | mant;
}

inline float half_to_float(uint16_t v) { return decode<5, 10, true>(v); }
inline uint16_t float_to_half(float f) { return static_cast<uint16_t>(encode<5, 10, true>(f)); }
inline float bf16_to_float(uint16_t v) { return decode<8, 7, true>(v); }
inline uint16_t float_to_bf16(float f) { return static_cast<uint16_t>(encode<8, 7, true>(f)); }
inline float e4m3_to_float(uint8_t v) { return decode<4, 3, false>(v); }
inline uint8_t float_to_e4m3(float f) { return static_cast<uint8_t>(encode<4, 3, false>(f)); }
inline float e5m2_to_float(uint8_t v) { return decode<5, 2, true>(v); }
inline uint8_t float_to_e5m2(float f) { return static_cast<uint8_t>(encode<5, 2, true>(f)); }

}  // namespace fp
```

- [ ] **Step 5: 12종 allreduce 테스트 작성** — `tests/test_dtypes.cc`

기준값은 MPI 없이 각 rank가 모든 rank의 입력을 다시 계산해 double로 리덕션한 뒤 그 dtype으로 변환한다. 입력은 모든 부분합·부분곱이 해당 dtype에서 정확히 표현되도록 고른다(합: 0/1 → 최대 8, 곱: 1/2 → 최대 256, max/min: −3..3, 부호 없는 형은 0..6). int8 곱 256은 0으로 감기고, 기준값도 `int64 → int8` 변환으로 같게 감긴다.

```cpp
#include <cstdint>
#include <cstring>
#include <vector>

#include "check.h"
#include "fp.h"
#include "hxccl.h"
#include "reduce.h"
#include "select.h"

namespace {
int rank_, n_;

double input(int r, size_t i, hxcclRedOp_t op, bool is_signed) {
  if (op == hxcclSum) return static_cast<double>((r + i) % 2);
  if (op == hxcclProd) return 1.0 + static_cast<double>((r + i) % 2);
  const int v = static_cast<int>((r * 3 + i) % 7);
  return is_signed ? v - 3 : v;
}

void store(double v, hxcclDataType_t dt, void* out) {
  switch (dt) {
    case hxcclInt8: { int8_t x = static_cast<int8_t>(static_cast<int64_t>(v)); std::memcpy(out, &x, 1); break; }
    case hxcclUint8: { uint8_t x = static_cast<uint8_t>(static_cast<int64_t>(v)); std::memcpy(out, &x, 1); break; }
    case hxcclInt32: { int32_t x = static_cast<int32_t>(v); std::memcpy(out, &x, 4); break; }
    case hxcclUint32: { uint32_t x = static_cast<uint32_t>(v); std::memcpy(out, &x, 4); break; }
    case hxcclInt64: { int64_t x = static_cast<int64_t>(v); std::memcpy(out, &x, 8); break; }
    case hxcclUint64: { uint64_t x = static_cast<uint64_t>(v); std::memcpy(out, &x, 8); break; }
    case hxcclFloat16: { uint16_t x = fp::float_to_half(static_cast<float>(v)); std::memcpy(out, &x, 2); break; }
    case hxcclBfloat16: { uint16_t x = fp::float_to_bf16(static_cast<float>(v)); std::memcpy(out, &x, 2); break; }
    case hxcclFloat8e4m3: { uint8_t x = fp::float_to_e4m3(static_cast<float>(v)); std::memcpy(out, &x, 1); break; }
    case hxcclFloat8e5m2: { uint8_t x = fp::float_to_e5m2(static_cast<float>(v)); std::memcpy(out, &x, 1); break; }
    case hxcclFloat32: { float x = static_cast<float>(v); std::memcpy(out, &x, 4); break; }
    case hxcclFloat64: std::memcpy(out, &v, 8); break;
    default: break;
  }
}

void check(hxcclComm_t c, Algo algo, hxcclDataType_t dt, hxcclRedOp_t op, size_t count) {
  const size_t es = dtype_size(dt);
  const bool sgn = dt != hxcclUint8 && dt != hxcclUint32 && dt != hxcclUint64;
  std::vector<char> in(count * es), got(count * es), want(count * es);
  for (size_t i = 0; i < count; ++i) {
    store(input(rank_, i, op, sgn), dt, in.data() + i * es);
    double acc = input(0, i, op, sgn);
    for (int r = 1; r < n_; ++r) {
      const double x = input(r, i, op, sgn);
      acc = op == hxcclSum ? acc + x : op == hxcclProd ? acc * x : op == hxcclMax ? std::max(acc, x) : std::min(acc, x);
    }
    store(acc, dt, want.data() + i * es);
  }
  CHECK(allreduce_with(c, algo, in.data(), got.data(), count, dt, op) == hxcclSuccess);
  if (got != want)
    std::fprintf(stderr, "dtype %d op %d algo %d count %zu\n", dt, op, static_cast<int>(algo), count);
  CHECK(got == want);
}
}  // namespace

int main(int argc, char** argv) {
  MPI_Init(&argc, &argv);
  MPI_Comm_rank(MPI_COMM_WORLD, &rank_);
  MPI_Comm_size(MPI_COMM_WORLD, &n_);
  hxcclComm_t c;
  CHECK(hxcclCommInitMPI(&c, MPI_COMM_WORLD) == hxcclSuccess);
  for (int d = 0; d < hxcclNumTypes; ++d)
```

```cpp
#include "reduce.h"

namespace {

template <typename T>
void apply(T* d, const T* s, size_t n, hxcclRedOp_t op) {
  // integer sum/prod wrap like the GPU does; unsigned math keeps signed overflow defined
  using U = std::conditional_t<std::is_integral_v<T>, std::make_unsigned_t<T>, T>;
  switch (op) {
    case hxcclSum:
      for (size_t i = 0; i < n; ++i) d[i] = static_cast<T>(static_cast<U>(d[i]) + static_cast<U>(s[i]));
      break;
    case hxcclProd:
      for (size_t i = 0; i < n; ++i) d[i] = static_cast<T>(static_cast<U>(d[i]) * static_cast<U>(s[i]));
      break;
    case hxcclMax:
      for (size_t i = 0; i < n; ++i) d[i] = std::max(d[i], s[i]);
      break;
    case hxcclMin:
      for (size_t i = 0; i < n; ++i) d[i] = std::min(d[i], s[i]);
      break;
    default:
      break;
  }
}

// fp16/bf16/fp8: widen to float, reduce, round back (NCCL does the same, fp8 via half).
template <typename S, float (*ToF)(S), S (*FromF)(float)>
void apply_small(S* d, const S* s, size_t n, hxcclRedOp_t op) {
  for (size_t i = 0; i < n; ++i) {
    const float a = ToF(d[i]), b = ToF(s[i]);
    const float r = op == hxcclSum ? a + b : op == hxcclProd ? a * b
                    : op == hxcclMax ? std::fmax(a, b) : std::fmin(a, b);
    d[i] = FromF(r);
  }
}

}  // namespace

size_t dtype_size(hxcclDataType_t dt) {
  switch (dt) {
    case hxcclInt8: case hxcclUint8: case hxcclFloat8e4m3: case hxcclFloat8e5m2: return 1;
    case hxcclFloat16: case hxcclBfloat16: return 2;
    case hxcclInt32: case hxcclUint32: case hxcclFloat32: return 4;
    case hxcclInt64: case hxcclUint64: case hxcclFloat64: return 8;
    default: return 0;
  }
}

bool is_integer(hxcclDataType_t dt) { return dt >= hxcclInt8 && dt <= hxcclUint64; }

bool valid_op(hxcclRedOp_t op) { return op >= hxcclSum && op <= hxcclMin; }

void reduce_into(void* dst, const void* src, size_t n, hxcclDataType_t dt, hxcclRedOp_t op) {
#define HX_CASE(DT, T) \
  case DT: apply(static_cast<T*>(dst), static_cast<const T*>(src), n, op); break;
#define HX_SMALL(DT, S, TO, FROM) \
  case DT: apply_small<S, TO, FROM>(static_cast<S*>(dst), static_cast<const S*>(src), n, op); break;
  switch (dt) {
    HX_CASE(hxcclInt8, int8_t)
    HX_CASE(hxcclUint8, uint8_t)
    HX_CASE(hxcclInt32, int32_t)
    HX_CASE(hxcclUint32, uint32_t)
    HX_CASE(hxcclInt64, int64_t)
    HX_CASE(hxcclUint64, uint64_t)
    HX_CASE(hxcclFloat32, float)
    HX_CASE(hxcclFloat64, double)
    HX_SMALL(hxcclFloat16, uint16_t, fp::half_to_float, fp::float_to_half)
    HX_SMALL(hxcclBfloat16, uint16_t, fp::bf16_to_float, fp::float_to_bf16)
    HX_SMALL(hxcclFloat8e4m3, uint8_t, fp::e4m3_to_float, fp::float_to_e4m3)
    HX_SMALL(hxcclFloat8e5m2, uint8_t, fp::e5m2_to_float, fp::float_to_e5m2)
    default: break;
```

```cpp
void store_scalar(double v, hxcclDataType_t dt, void* out) {
  const float f = static_cast<float>(v);
  switch (dt) {
    case hxcclFloat64: std::memcpy(out, &v, 8); break;
    case hxcclFloat32: std::memcpy(out, &f, 4); break;
    case hxcclFloat16: { uint16_t h = fp::float_to_half(f); std::memcpy(out, &h, 2); break; }
    case hxcclBfloat16: { uint16_t h = fp::float_to_bf16(f); std::memcpy(out, &h, 2); break; }
    case hxcclFloat8e4m3: { uint8_t h = fp::float_to_e4m3(f); std::memcpy(out, &h, 1); break; }
    case hxcclFloat8e5m2: { uint8_t h = fp::float_to_e5m2(f); std::memcpy(out, &h, 1); break; }
    default: break;
  }
}

void scale_into(void* dst, const void* src, size_t n, hxcclDataType_t dt, const void* scalar) {
  const size_t es = dtype_size(dt);
  // dst = src, then dst *= scalar elementwise via the Prod reduction
  std::memcpy(dst, src, n * es);
  std::vector<char> k(n * es);
  for (size_t i = 0; i < n; ++i) std::memcpy(k.data() + i * es, scalar, es);
  reduce_into(dst, k.data(), n, dt, hxcclProd);
}

void divide_trunc(void* buf, size_t n, hxcclDataType_t dt, int divisor) {
#define HX_DIV(DT, T) \
  case DT: for (size_t i = 0; i < n; ++i) static_cast<T*>(buf)[i] = static_cast<T>(static_cast<T*>(buf)[i] / static_cast<T>(divisor)); break;
  switch (dt) {
    HX_DIV(hxcclInt8, int8_t)
    HX_DIV(hxcclUint8, uint8_t)
    HX_DIV(hxcclInt32, int32_t)
    HX_DIV(hxcclUint32, uint32_t)
    HX_DIV(hxcclInt64, int64_t)
    HX_DIV(hxcclUint64, uint64_t)
    default: break;
  }
#undef HX_DIV
}
```

```cpp
hxcclResult_t plan_op(hxcclComm* c, hxcclRedOp_t op, hxcclDataType_t dt, OpPlan* p) {
  *p = OpPlan{};
  if (valid_op(op)) {
    p->base = op;
    return hxcclSuccess;
  }
  if (op == hxcclAvg) {  // NCCL enqueue.cc: ints sum then divide, floats premultiply by 1/n
    if (is_integer(dt)) {
      p->postdiv = true;
      p->divisor = c->nranks;
    } else {
      p->premul = true;
      store_scalar(1.0 / c->nranks, dt, p->scalar);
    }
    return hxcclSuccess;
  }
  const long ix = static_cast<long>(op) - hxcclNumOps;
  if (ix < 0 || ix >= static_cast<long>(c->user_ops.size()) || !c->user_ops[ix].live)
    return comm_fail(c, hxcclInvalidArgument, "unknown op");
  const UserOp& u = c->user_ops[ix];
  if (u.dt != dt)
    return comm_fail(c, hxcclInvalidArgument, "PreMulSum op was created for another datatype");
  p->premul = true;
  std::memcpy(p->scalar, u.by_ptr ? u.ptr : u.value, dtype_size(dt));
  return hxcclSuccess;
}
```

```cpp
                             hxcclRedOp_t op, hxcclComm_t c, hxcclStream_t) {

  OpPlan p;

  const size_t es = dtype_size(dt);
```

```cpp
hxcclResult_t hxcclRedOpCreatePreMulSum(hxcclRedOp_t* op, void* scalar, hxcclDataType_t dt,
                                        hxcclScalarResidence_t res, hxcclComm_t c) {
  if (!c || !op || !scalar) return hxcclInvalidArgument;
  if (dtype_size(dt) == 0) return comm_fail(c, hxcclInvalidArgument, "unknown datatype");
  if (res != hxcclScalarDevice && res != hxcclScalarHostImmediate)
    return comm_fail(c, hxcclInvalidArgument, "unknown scalar residence");
  size_t ix = 0;
  while (ix < c->user_ops.size() && c->user_ops[ix].live) ++ix;
  if (ix == c->user_ops.size()) c->user_ops.emplace_back();
  UserOp& u = c->user_ops[ix];
  u = UserOp{};
  u.live = true;
  u.dt = dt;
  u.by_ptr = res == hxcclScalarDevice;  // CPU buffers: "device" means read at call time
  if (u.by_ptr) u.ptr = scalar;
  else std::memcpy(u.value, scalar, dtype_size(dt));
  *op = static_cast<hxcclRedOp_t>(hxcclNumOps + static_cast<int>(ix));
  return hxcclSuccess;
}

hxcclResult_t hxcclRedOpDestroy(hxcclRedOp_t op, hxcclComm_t c) {
  if (!c) return hxcclInvalidArgument;
  const long ix = static_cast<long>(op) - hxcclNumOps;
  if (ix < 0 || ix >= static_cast<long>(c->user_ops.size()) || !c->user_ops[ix].live)
    return comm_fail(c, hxcclInvalidArgument, "not a live PreMulSum op of this communicator");
  c->user_ops[ix].live = false;
  return hxcclSuccess;
}
```

```cpp
hxcclResult_t MpiTransport::waitall() {
  if (pending_.empty()) return hxcclSuccess;
  std::vector<MPI_Status> st(pending_.size());
  int rc = MPI_Waitall(static_cast<int>(pending_.size()), pending_.data(), st.data());
  pending_.clear();
  if (rc == MPI_ERR_IN_STATUS)
    for (const MPI_Status& s : st)
      if (s.MPI_ERROR != MPI_SUCCESS) return check(s.MPI_ERROR, "MPI_Waitall");
  return check(rc, "MPI_Waitall");
}

void MpiTransport::cancel_all() {
  for (MPI_Request& r : pending_) {
    MPI_Cancel(&r);
    MPI_Request_free(&r);
  }
  pending_.clear();
}
```

```cpp
#include "group.h"

#include <algorithm>
#include <utility>
#include <vector>

namespace {

struct P2p {
  hxcclComm* c;
  bool send;
  int peer;
  void* buf;
  size_t bytes;
};

struct Group {
  int depth = 0;
  std::vector<std::pair<hxcclComm*, std::function<hxcclResult_t()>>> colls;
  std::vector<P2p> p2p;
};
thread_local Group g;

// Post every send/recv before waiting on any, so pairs that send to each other finish.
hxcclResult_t run_p2p(const std::vector<P2p>& ops) {
  std::vector<hxcclComm*> comms;
  for (const P2p& o : ops) {
    if (std::find(comms.begin(), comms.end(), o.c) == comms.end()) comms.push_back(o.c);
    Transport& tr = *o.c->tr;
    hxcclResult_t r = o.send ? tr.isend(o.peer, o.buf, o.bytes, kP2pTag)
                             : tr.irecv(o.peer, o.buf, o.bytes, kP2pTag);
    if (r != hxcclSuccess) {
      for (hxcclComm* c : comms) c->tr->cancel_all();
      return comm_fail(o.c, r, tr.error);
    }
  }
  hxcclResult_t first = hxcclSuccess;
  for (hxcclComm* c : comms) {
    hxcclResult_t r = c->tr->waitall();
    if (r != hxcclSuccess && first == hxcclSuccess) first = comm_fail(c, r, c->tr->error);
  }
  return first;
}

}  // namespace

hxcclResult_t dispatch(hxcclComm* c, std::function<hxcclResult_t()> run) {
  if (g.depth == 0) return run();
  g.colls.emplace_back(c, std::move(run));
  return hxcclSuccess;
}

hxcclResult_t p2p(hxcclComm* c, bool is_send, int peer, void* buf, size_t bytes) {
  P2p op{c, is_send, peer, buf, bytes};
  if (g.depth == 0) return run_p2p({op});
  g.p2p.push_back(op);
  return hxcclSuccess;
}

extern "C" {

hxcclResult_t hxcclGroupStart(void) {
  ++g.depth;
  return hxcclSuccess;
}

hxcclResult_t hxcclGroupEnd(void) {
  if (g.depth == 0) return hxcclInvalidUsage;
  if (--g.depth > 0) return hxcclSuccess;
  auto colls = std::move(g.colls);
  auto ops = std::move(g.p2p);
  g.colls.clear();
  g.p2p.clear();
  for (auto& [c, run] : colls)  // collectives first, in call order (spec 5.2)
    if (hxcclResult_t r = run()) return r;
  return ops.empty() ? hxcclSuccess : run_p2p(ops);
}

}  // extern "C"
```

```cpp
#include <algorithm>
#include <cstring>
#include <vector>

#include "algo.h"
#include "reduce.h"

hxcclResult_t reduce_root(Transport& tr, int rank, int n, const void* send, void* recv,
                          size_t count, hxcclDataType_t dt, hxcclRedOp_t op, int root) {
  const size_t es = dtype_size(dt), bytes = count * es;
  std::vector<char> own;
  char* buf;
  if (rank == root) {
    buf = static_cast<char*>(recv);
    if (send != recv) std::memcpy(buf, send, bytes);
  } else {  // non-root recv may be NULL (NCCL), so partial sums live in a private copy
    own.assign(static_cast<const char*>(send), static_cast<const char*>(send) + bytes);
    buf = own.data();
  }
  if (n == 1 || count == 0) return hxcclSuccess;
  const int v = (rank - root + n) % n;
  auto real = [=](int x) { return (x + root) % n; };
  hxcclResult_t r;
  if (bytes < kChainMinBytes) {  // binomial tree, mirror of bcast
    std::vector<char> tmp(bytes);
    for (int mask = 1; mask < n; mask <<= 1) {
      if (v & mask) return tr.send(real(v - mask), buf, bytes);
      if (v + mask < n) {
        if ((r = tr.recv(real(v + mask), tmp.data(), bytes)) != hxcclSuccess) return r;
        reduce_into(buf, tmp.data(), count, dt, op);
      }
    }
    return hxcclSuccess;
  }
  // chain toward the root in kChainSegment pieces (a multiple of every dtype size)
  std::vector<char> tmp(kChainSegment);
  for (size_t off = 0; off < bytes; off += kChainSegment) {
    const size_t len = std::min(kChainSegment, bytes - off);
    if (v < n - 1) {
      if ((r = tr.recv(real(v + 1), tmp.data(), len)) != hxcclSuccess) return r;
      reduce_into(buf + off, tmp.data(), len / es, dt, op);
    }
    if (v > 0 && (r = tr.send(real(v - 1), buf + off, len)) != hxcclSuccess) return r;
  }
  return hxcclSuccess;
}

hxcclResult_t gather(Transport& tr, int rank, int n, const void* send, void* recv,
                     size_t per, int root) {
  if (per == 0) return hxcclSuccess;
  if (rank != root) return tr.send(root, send, per);
  char* out = static_cast<char*>(recv);
  if (send != out + root * per) std::memcpy(out + root * per, send, per);
  for (int i = 0; i < n; ++i)
    if (i != root)
      if (hxcclResult_t r = tr.irecv(i, out + i * per, per, kCollTag)) {
        tr.cancel_all();
        return r;
      }
  return tr.waitall();
}

hxcclResult_t scatter(Transport& tr, int rank, int n, const void* send, void* recv,
                      size_t per, int root) {
  if (per == 0) return hxcclSuccess;
  if (rank != root) return tr.recv(root, recv, per);
  const char* in = static_cast<const char*>(send);
  for (int i = 0; i < n; ++i)
    if (i != root)
      if (hxcclResult_t r = tr.isend(i, in + i * per, per, kCollTag)) {
        tr.cancel_all();
        return r;
      }
  if (recv != in + root * per) std::memcpy(recv, in + root * per, per);
  return tr.waitall();
}
```

```cpp
hxcclResult_t fail_if(hxcclComm* c, hxcclResult_t r) {
  return r == hxcclSuccess ? r : comm_fail(c, r, c->tr->error);
}
}  // namespace

extern "C" {

hxcclResult_t hxcclReduce(const void* send, void* recv, size_t count, hxcclDataType_t dt,
                          hxcclRedOp_t op, int root, hxcclComm_t c, hxcclStream_t) {
  if (!c) return hxcclInvalidArgument;
  if (hxcclResult_t r = check_root(c, root)) return r;
  if (hxcclResult_t r = check_args(c, send, c->rank == root ? recv : send, count, 1, dt)) return r;
  OpPlan p;
  if (hxcclResult_t r = plan_op(c, op, dt, &p)) return r;
  const size_t bytes = count * dtype_size(dt);
  if (hxcclResult_t r = check_message(c, bytes < kChainMinBytes ? bytes : kChainSegment)) return r;
  return dispatch(c, [=] {
    std::vector<char> tmp;
    const void* in = premul_input(p, send, count, dt, tmp);
    hxcclResult_t r = fail_if(c, reduce_root(*c->tr, c->rank, c->nranks, in, recv, count, dt, p.base, root));
    if (r == hxcclSuccess && c->rank == root) finish_op(p, recv, count, dt);
    return r;
  });
}

hxcclResult_t hxcclBcast(void* buff, size_t count, hxcclDataType_t dt, int root, hxcclComm_t c,
                         hxcclStream_t s) {
  return hxcclBroadcast(buff, buff, count, dt, root, c, s);
}

hxcclResult_t hxcclGather(const void* send, void* recv, size_t count, hxcclDataType_t dt,
                          int root, hxcclComm_t c, hxcclStream_t) {
  if (!c) return hxcclInvalidArgument;
  if (hxcclResult_t r = check_root(c, root)) return r;
  if (hxcclResult_t r = check_args(c, send, c->rank == root ? recv : send, count, c->nranks, dt))
    return r;
  const size_t per = count * dtype_size(dt);
  if (hxcclResult_t r = check_message(c, per)) return r;
  return dispatch(c, [=] { return fail_if(c, gather(*c->tr, c->rank, c->nranks, send, recv, per, root)); });
}

hxcclResult_t hxcclScatter(const void* send, void* recv, size_t count, hxcclDataType_t dt,
                           int root, hxcclComm_t c, hxcclStream_t) {
  if (!c) return hxcclInvalidArgument;
  if (hxcclResult_t r = check_root(c, root)) return r;
  if (hxcclResult_t r = check_args(c, c->rank == root ? send : recv, recv, count, c->nranks, dt))
    return r;
  const size_t per = count * dtype_size(dt);
  if (hxcclResult_t r = check_message(c, per)) return r;
  return dispatch(c, [=] { return fail_if(c, scatter(*c->tr, c->rank, c->nranks, send, recv, per, root)); });
}
```

```cpp
int rank_, n_;

void alltoall(hxcclComm_t c, size_t count, bool rccl_name) {
  std::vector<int32_t> in(count * n_), want(count * n_), got(count * n_, -1);
  for (size_t i = 0; i < in.size(); ++i) in[i] = rank_ * 1000000 + static_cast<int32_t>(i);

void alltoallv(hxcclComm_t c) {
  std::vector<size_t> sc(n_), sd(n_), rc(n_), rd(n_);
```

```cpp
#include <cstring>

#include "algo.h"

// Pairwise exchange: at step s every rank sends to r+s and receives from r-s.
hxcclResult_t alltoallv(Transport& tr, int rank, int n, const char* send, const size_t* sbytes,
                        const size_t* soff, char* recv, const size_t* rbytes,
                        const size_t* roff) {
  if (rbytes[rank]) std::memcpy(recv + roff[rank], send + soff[rank], rbytes[rank]);
  for (int s = 1; s < n; ++s) {
    const int to = (rank + s) % n, from = (rank - s + n) % n;
    if (hxcclResult_t r = tr.sendrecv(to, send + soff[to], sbytes[to], from,
                                      recv + roff[from], rbytes[from]))
      return r;
  }
  return hxcclSuccess;
}
```

- [ ] **Step 4: API** — `src/api_alltoall.cc`

AllToAllv는 rank마다 블록 크기가 달라 혼자 거절하면 다른 rank가 멈춘다. 그래서 로컬 검사 결과를 `MPI_Allreduce(MAX)`로 합의한 뒤 다 같이 거절한다(스펙 5.3). 이 합의 통신은 Transport를 거치지 않아 통계에 들어가지 않는다.

```cpp
#include <cstdint>
#include <vector>

#include "algo.h"
#include "checks.h"
#include "comm.h"
#include "group.h"
#include "reduce.h"

namespace {
hxcclResult_t fail_if(hxcclComm* c, hxcclResult_t r) {
  return r == hxcclSuccess ? r : comm_fail(c, r, c->tr->error);
}

// 0 ok, 1 invalid argument, 2 a block exceeds INT_MAX bytes
int local_v_status(const hxcclComm* c, const void* send, const size_t* sc, const size_t* sd,
                   const void* recv, const size_t* rc, const size_t* rd, size_t es) {
  if (es == 0 || !sc || !sd || !rc || !rd) return 1;
  int status = 0;
  for (int j = 0; j < c->nranks; ++j) {
    for (size_t cnt : {sc[j], rc[j]}) {
      if (cnt > SIZE_MAX / es) return 1;
      if (cnt * es > static_cast<size_t>(INT_MAX)) status = 2;
    }
    if (sd[j] > SIZE_MAX / es - sc[j] || rd[j] > SIZE_MAX / es - rc[j]) return 1;
    if ((sc[j] && !send) || (rc[j] && !recv)) return 1;
  }
  return status;
}
}  // namespace

extern "C" {

hxcclResult_t hxcclAllToAllv(const void* send, const size_t sc[], const size_t sd[], void* recv,
                             const size_t rc[], const size_t rd[], hxcclDataType_t dt,
                             hxcclComm_t c, hxcclStream_t) {
  if (!c) return hxcclInvalidArgument;
  const size_t es = dtype_size(dt);
  int status = local_v_status(c, send, sc, sd, recv, rc, rd, es);
  MPI_Allreduce(MPI_IN_PLACE, &status, 1, MPI_INT, MPI_MAX, c->mpi);
  if (status == 1) return comm_fail(c, hxcclInvalidArgument, "invalid AllToAllv arguments on some rank");
  if (status == 2)
    return comm_fail(c, hxcclInvalidArgument,
                     "a single message would exceed INT_MAX bytes (MPI transport limit)");
  std::vector<size_t> sb(c->nranks), so(c->nranks), rb(c->nranks), ro(c->nranks);
  for (int j = 0; j < c->nranks; ++j) {
    sb[j] = sc[j] * es; so[j] = sd[j] * es; rb[j] = rc[j] * es; ro[j] = rd[j] * es;
  }
  return dispatch(c, [=] {
    return fail_if(c, alltoallv(*c->tr, c->rank, c->nranks, static_cast<const char*>(send),
                                sb.data(), so.data(), static_cast<char*>(recv), rb.data(), ro.data()));
  });
}

hxcclResult_t hxcclAlltoAll(const void* send, void* recv, size_t count, hxcclDataType_t dt,
                            hxcclComm_t c, hxcclStream_t) {
  if (hxcclResult_t r = check_args(c, send, recv, count, c ? c->nranks : 1, dt)) return r;
  const size_t per = count * dtype_size(dt);
  if (hxcclResult_t r = check_message(c, per)) return r;
  if (count == 0) return hxcclSuccess;
  std::vector<size_t> b(c->nranks, per), off(c->nranks);
  for (int j = 0; j < c->nranks; ++j) off[j] = static_cast<size_t>(j) * per;
  return dispatch(c, [=] {
    return fail_if(c, alltoallv(*c->tr, c->rank, c->nranks, static_cast<const char*>(send),
                                b.data(), off.data(), static_cast<char*>(recv), b.data(), off.data()));
  });
}

hxcclResult_t hxcclAllToAll(const void* send, void* recv, size_t count, hxcclDataType_t dt,
                            hxcclComm_t c, hxcclStream_t s) {
  return hxcclAlltoAll(send, recv, count, dt, c, s);
}

}  // extern "C"
```

```cpp
namespace {
int rank_, n_;

struct FailingTransport : Transport {
  hxcclResult_t fail() { error = "injected"; return hxcclSystemError; }
  hxcclResult_t sendrecv(int, const void*, size_t, int, void*, size_t) override { return fail(); }
  hxcclResult_t send(int, const void*, size_t) override { return fail(); }
  hxcclResult_t recv(int, void*, size_t) override { return fail(); }
  hxcclResult_t isend(int, const void*, size_t, int) override { return fail(); }

void split_parity(hxcclComm_t c) {
  hxcclComm_t sub = nullptr;
  CHECK(hxcclCommSplit(c, rank_ % 2, -rank_, &sub, nullptr) == hxcclSuccess);
  int sn = 0, sr = 0;
  hxcclCommCount(sub, &sn);
  hxcclCommUserRank(sub, &sr);
  CHECK(sn == (n_ + 1 - rank_ % 2) / 2);
  CHECK(sr == sn - 1 - rank_ / 2);  // key = -rank reverses the order
  int32_t x = rank_, sum = 0, want = 0;
  for (int r = rank_ % 2; r < n_; r += 2) want += r;
  CHECK(hxcclAllReduce(&x, &sum, 1, hxcclInt32, hxcclSum, sub, nullptr) == hxcclSuccess);
  CHECK(sum == want);
  hxcclCommDestroy(sub);
}

void split_nocolor(hxcclComm_t c) {

void group_two_comms(hxcclComm_t c) {

  hxcclCommSplit(c, rank_ % 2, rank_, &sub, nullptr);
  int32_t x = 1, all = 0, part = 0;
  hxcclGroupStart();
  hxcclAllReduce(&x, &part, 1, hxcclInt32, hxcclSum, sub, nullptr);

void shrink(hxcclComm_t c) {

  if (rank_ == 0 || rank_ == n_ - 1) return;  // excluded ranks do not call (NCCL)
  hxcclComm_t s = nullptr;

  int sn = 0, sr = 0;
  hxcclCommCount(s, &sn);
  hxcclCommUserRank(s, &sr);
  CHECK(sn == n_ - 2 && sr == rank_ - 1);
  int32_t x = 1, sum = 0;
  CHECK(hxcclAllReduce(&x, &sum, 1, hxcclInt32, hxcclSum, s, nullptr) == hxcclSuccess);
  CHECK(sum == n_ - 2);
  hxcclCommDestroy(s);
}

void shrink_errors(hxcclComm_t c) {
  hxcclComm_t s = nullptr;
  int self = rank_, out = n_, dup[2] = {0, 0};
  CHECK(hxcclCommShrink(c, &self, 1, &s, nullptr, 0) == hxcclInvalidUsage);
  CHECK(hxcclCommShrink(c, &out, 1, &s, nullptr, 0) == hxcclInvalidArgument);
  CHECK(hxcclCommShrink(c, dup, 2, &s, nullptr, 0) == hxcclInvalidArgument);
  CHECK(hxcclCommShrink(c, nullptr, 0, &s, nullptr, 1) == hxcclInvalidArgument);

// Abort is local: odd ranks abort, even ranks never touch the comm, the barrier still passes
void abort_is_local() {
  hxcclComm_t a;
  CHECK(hxcclCommInitMPI(&a, MPI_COMM_WORLD) == hxcclSuccess);
  if (rank_ % 2) CHECK(hxcclCommAbort(a) == hxcclSuccess);

}

void finalize_then_use() {
  hxcclComm_t f;
  CHECK(hxcclCommInitMPI(&f, MPI_COMM_WORLD) == hxcclSuccess);
  CHECK(hxcclCommFinalize(f) == hxcclSuccess);
  int32_t x = 1;
  CHECK(hxcclAllReduce(&x, &x, 1, hxcclInt32, hxcclSum, f, nullptr) == hxcclInvalidUsage);
  CHECK(hxcclSend(&x, 1, hxcclInt32, 0, f, nullptr) == hxcclInvalidUsage);
  CHECK(hxcclCommDestroy(f) == hxcclSuccess);
}

void async_error() {
  hxcclComm_t f;
  CHECK(hxcclCommInitMPI(&f, MPI_COMM_WORLD) == hxcclSuccess);
  hxcclResult_t e = hxcclInternalError;
  CHECK(hxcclCommGetAsyncError(f, &e) == hxcclSuccess && e == hxcclSuccess);
  float x = 0;
  hxcclAllReduce(&x, &x, 1, static_cast<hxcclDataType_t>(42), hxcclSum, f, nullptr);
  hxcclCommGetAsyncError(f, &e);
  CHECK(e == hxcclSuccess);  // argument errors are not communicator state

    f->tr = std::make_unique<FailingTransport>();
    hxcclAllReduce(&x, &x, 1, hxcclFloat32, hxcclSum, f, nullptr);
    hxcclCommGetAsyncError(f, &e);
    CHECK(e == hxcclSystemError);

  hxcclCommDestroy(f);
}

  CHECK(hxcclCommInitRankConfig(&s, n_, id, rank_, &nb) == hxcclInvalidArgument);

}
```

```cpp
                                      hxcclConfig_t* config) {
  if (hxcclResult_t r = check_config(config)) return r;

hxcclResult_t hxcclCommSplit(hxcclComm_t comm, int color, int key, hxcclComm_t* newcomm,
                             hxcclConfig_t* config) {
  if (!comm || !newcomm || (color < 0 && color != HXCCL_SPLIT_NOCOLOR)) return hxcclInvalidArgument;
  if (hxcclResult_t r = check_config(config)) return comm_fail(comm, r, "unsupported config");
  MPI_Comm sub;
  if (MPI_Comm_split(comm->mpi, color == HXCCL_SPLIT_NOCOLOR ? MPI_UNDEFINED : color, key, &sub) !=
      MPI_SUCCESS)
    return comm_fail(comm, hxcclSystemError, "MPI_Comm_split failed");
  if (sub == MPI_COMM_NULL) {
    *newcomm = nullptr;
    return hxcclSuccess;
  }
  return make_comm(newcomm, sub);
}

hxcclResult_t hxcclCommShrink(hxcclComm_t comm, int* excl, int nexcl, hxcclComm_t* newcomm,
                              hxcclConfig_t* config, int flags) {
  if (!comm || !newcomm || nexcl < 0 || (nexcl > 0 && !excl)) return hxcclInvalidArgument;
  if (flags != 0) return comm_fail(comm, hxcclInvalidArgument, "shrinkFlags other than 0 arrive in E");
  if (hxcclResult_t r = check_config(config)) return comm_fail(comm, r, "unsupported config");
  std::vector<bool> seen(comm->nranks, false);
  for (int i = 0; i < nexcl; ++i) {
    if (excl[i] < 0 || excl[i] >= comm->nranks || seen[excl[i]])
      return comm_fail(comm, hxcclInvalidArgument, "excluded rank out of range or repeated");
    seen[excl[i]] = true;
  }
  if (seen[comm->rank]) return comm_fail(comm, hxcclInvalidUsage, "excluded ranks must not call Shrink");
  MPI_Group all, kept;
  MPI_Comm sub;
  MPI_Comm_group(comm->mpi, &all);
  MPI_Group_excl(all, nexcl, excl, &kept);
  const int rc = MPI_Comm_create_group(comm->mpi, kept, /*tag=*/7, &sub);
  MPI_Group_free(&kept);
  MPI_Group_free(&all);
  if (rc != MPI_SUCCESS) return comm_fail(comm, hxcclSystemError, "MPI_Comm_create_group failed");
  return make_comm(newcomm, sub);
}

hxcclResult_t hxcclCommAbort(hxcclComm_t comm) {
  if (!comm) return hxcclInvalidArgument;
  // local teardown only: MPI_Comm_free may need peers, so the MPI handle is leaked on purpose
  comm->tr.reset();
  delete comm;
  return hxcclSuccess;
}

hxcclResult_t hxcclCommFinalize(hxcclComm_t comm) {
  if (!comm) return hxcclInvalidArgument;
  comm->finalized = true;  // A2 runs every call to completion, nothing is left in flight
  return hxcclSuccess;
}

hxcclResult_t hxcclCommGetAsyncError(hxcclComm_t comm, hxcclResult_t* err) {
  if (!comm || !err) return hxcclInvalidArgument;
  *err = comm->async_error;
  return hxcclSuccess;
}
```
